        <div class="flex flex-wrap gap-2">
          <button onclick="setAspectRatio(NaN)" class="px-3 py-2 bg-white active:bg-slate-200 rounded-lg shadow-sm border border-slate-200 font-medium text-sm">フリー</button>
          <button onclick="setAspectRatio(1)" class="px-3 py-2 bg-white active:bg-slate-200 rounded-lg shadow-sm border border-slate-200 font-medium text-sm">1:1</button>
          <button onclick="setAspectRatio(4/3)" class="px-3 py-2 bg-white active:bg-slate-200 rounded-lg shadow-sm border border-slate-200 font-medium text-sm">4:3</button>
          <button onclick="setAspectRatio(1/1.414)" class="px-3 py-2 bg-white active:bg-slate-200 rounded-lg shadow-sm border border-slate-200 font-medium text-sm">A4縦</button>
          <button onclick="rotateImage(90)" class="px-3 py-2 bg-white active:bg-slate-200 rounded-lg shadow-sm border border-slate-200 font-medium text-sm">🔄 90°回転</button>

      <div id="croppedList" class="flex flex-wrap gap-3 mb-4"></div>

  <script>
    const GOOGLE_CLIENT_ID = 'YOUR_GOOGLE_CLIENT_ID.apps.googleusercontent.com';

    let cropper = null;
    let selectedImageIndex = 0;
    let imagesList = [];
    let croppedImages = [];

    // ローカル／iPad写真ライブラリからの読み込み
    function loadLocalFiles(e) {
      const files = Array.from(e.target.files);
      if (files.length === 0) return;

      let loadedCount = 0;
      files.forEach(file => {
        const reader = new FileReader();
        reader.onload = (evt) => {
          imagesList.push(evt.target.result);
          loadedCount++;
          if (loadedCount === files.length) {
            renderThumbnails();
            selectImageForCrop(imagesList.length - files.length);
          }
        };
        reader.readAsDataURL(file);
      });
      e.target.value = ''; // リセット
    }

    function renderThumbnails() {
      const container = document.getElementById('thumbContainer');
      container.classList.remove('hidden');
      container.innerHTML = imagesList.map((src, idx) => `
        <img src="${src}" class="w-16 h-16 object-cover rounded-lg cursor-pointer border-2 ${idx === selectedImageIndex ? 'border-blue-500 scale-105' : 'border-transparent'} transition-transform" onclick="selectImageForCrop(${idx})">
      `).join('');
    }

    function selectImageForCrop(index) {
      selectedImageIndex = index;
      renderThumbnails();
      
      const cropContainer = document.getElementById('cropContainer');
      const cropImage = document.getElementById('cropImage');
      cropContainer.classList.remove('hidden');

      cropImage.src = imagesList[index];

      if (cropper) cropper.destroy();

      // iPad向けタッチ操作最適化オプション
      cropper = new Cropper(cropImage, {
        viewMode: 1,
        autoCropArea: 0.85,
        responsive: true,
        rotatable: true,
        scalable: true,
        zoomable: true,
        touchDragZoom: true,
        mouseWheelZoom: true,
      });
    }

    function setAspectRatio(ratio) {
      if (cropper) cropper.setAspectRatio(ratio);
    }

    function rotateImage(deg) {
      if (cropper) cropper.rotate(deg);
    }

    function saveCroppedImage() {
      if (!cropper) return;
      const canvas = cropper.getCroppedCanvas({ maxWidth: 2048, maxHeight: 2048 });
      const croppedDataUrl = canvas.toDataURL('image/jpeg', 0.88);
      
      croppedImages.push(croppedDataUrl);
      renderCroppedList();
    }

    function renderCroppedList() {
      const area = document.getElementById('pdfExportArea');
      const list = document.getElementById('croppedList');
      const count = document.getElementById('croppedCount');

      area.classList.remove('hidden');
      count.textContent = croppedImages.length;

      list.innerHTML = croppedImages.map((src, idx) => `
        <div class="relative w-20 h-20 border rounded-lg overflow-hidden shadow-sm bg-slate-200">
          <img src="${src}" class="w-full h-full object-cover">
          <button onclick="removeCropped(${idx})" class="absolute top-1 right-1 bg-red-600 text-white rounded-full w-5 h-5 text-xs flex items-center justify-center font-bold">✕</button>
        </div>
      `).join('');
    }

    function removeCropped(index) {
      croppedImages.splice(index, 1);
      renderCroppedList();
    }

    // PDF生成処理（iPad Safari対応）
    function generatePDF() {
      if (croppedImages.length === 0) return;

      const { jsPDF } = window.jspdf;
      const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });
      const pageWidth = 210;
      const pageHeight = 297;
      const margin = 10;

      croppedImages.forEach((imgData, idx) => {
        if (idx > 0) doc.addPage();

        const imgProps = doc.getImageProperties(imgData);
        const maxW = pageWidth - margin * 2;
        const maxH = pageHeight - margin * 2;
        
        let width = maxW;
        let height = (imgProps.height * width) / imgProps.width;

        if (height > maxH) {
          height = maxH;
          width = (imgProps.width * height) / imgProps.height;
        }

        const x = (pageWidth - width) / 2;
        const y = (pageHeight - height) / 2;

        doc.addImage(imgData, 'JPEG', x, y, width, height);
      });

      // iPad SafariでPDFを保存/表示
      doc.save('cropped-photos.pdf');
    }

    // Google API 認証枠組
    let tokenClient;
    function initGoogleAuth() {
      if (typeof google === 'undefined') return;
      tokenClient = google.accounts.oauth2.initTokenClient({
        client_id: GOOGLE_CLIENT_ID,
        scope: 'https://www.googleapis.com/auth/photospicker.mediaitems.readonly',
        callback: async (res) => { if (res.access_token) fetchGooglePhotos(res.access_token); }
      });
    }
    function handleAuth() {
      if (!GOOGLE_CLIENT_ID || GOOGLE_CLIENT_ID.startsWith('YOUR_')) {
        alert('iPadの「写真」ボタンから直接写真を選ぶか、Google Client IDを設定してください。');
        return;
      }
      if (!tokenClient) initGoogleAuth();
      tokenClient.requestAccessToken();
    }
    async function fetchGooglePhotos(token) {
      const res = await fetch('https://photospicker.googleapis.com/v1/sessions', {
        method: 'POST',
        headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' }
      });
      const session = await res.json();
      if (session.pickerUri) window.open(session.pickerUri, '_blank');
    }
  </script>
